## 七、核心算法

### 7.1 TXT/MD 解析算法

**解析规则（宽松容错）**：
- 题目分隔：`---` 或 `## 题目N`
- 题干：首行非选项/答案/解析的文本
- 选项：匹配 `A.` `B.` `C.` `D.` `对.` `错.` 开头的行
- 答案：匹配 `答案：` 或 `ANS:` 关键字后的内容
- 解析：匹配 `解析：` 或 `EXP:` 关键字后的内容

**解析流程**：
```
用户粘贴文本
    ↓
前端提取纯文本
    ↓
判断格式：
    ├── MD格式 → 正则解析
    └── TXT格式 → 正则解析（按约定分隔符）
        ↓
解析失败 → 调用AI API解析（可选）
        ↓
结构化JSON预览
    ↓
用户编辑/删除/确认
    ↓
选择存储：
    ├── 本地 → IndexedDB
    └── 云端 → 需登录 → Supabase
```

### 7.2 组卷算法

```javascript
function generateQuiz(bankId, questionCount, options = {}) {
  // 1. 获取题库所有题目
  const pool = await getQuestionsByBank(bankId);

  // 2. 过滤（如按难度、标签）
  let filtered = pool;
  if (options.difficulty) {
    filtered = filtered.filter(q => q.difficulty === options.difficulty);
  }
  if (options.excludeIds) {
    filtered = filtered.filter(q => !options.excludeIds.includes(q.id));
  }

  // 3. 随机洗牌（Fisher-Yates）
  const shuffled = shuffle([...filtered]);

  // 4. 截取题量
  const selected = shuffled.slice(0, Math.min(questionCount, shuffled.length));

  // 5. 选项乱序 + 记录新正确答案位置
  return selected.map(q => {
    const { options: shuffledOpts, newCorrectIndex } = shuffleOptions(q.options, q.correctIndex);
    return {
      ...q,
      displayOptions: shuffledOpts,
      displayCorrectIndex: newCorrectIndex,
      userAnswer: null,
      isMarked: false,
      showExplanation: options.feedbackMode === 'instant'
    };
  });
}

function shuffleOptions(options, correctIndex) {
  const indexed = options.map((opt, i) => ({ ...opt, originalIndex: i }));
  const shuffled = shuffle(indexed);

  const newCorrectIndex = Array.isArray(correctIndex)
    ? correctIndex.map(ci => shuffled.findIndex(s => s.originalIndex === ci))
    : shuffled.findIndex(s => s.originalIndex === correctIndex);

  return {
    options: shuffled.map(({ originalIndex, ...opt }) => opt),
    newCorrectIndex
  };
}
```

### 7.3 判分算法

#### 7.3.1 考试型计分

```javascript
function calculateExamScore(answers, questions) {
  let correctCount = 0;
  const maxScore = 100;

  for (let i = 0; i < answers.length; i++) {
    if (isAnswerCorrect(answers[i], questions[i])) {
      correctCount++;
    }
  }

  const perQuestionScore = maxScore / answers.length;
  const totalScore = Math.round(correctCount * perQuestionScore);

  return {
    score: totalScore,
    correctCount,
    total: answers.length
  };
}
```

#### 7.3.2 游戏型计分（增强版）

```javascript
function calculateGameScore(answers, questions, config = {}) {
  let totalScore = 0;
  let streak = 0;
  let maxStreak = 0;

  const baseScore = config.baseScore || 10;
  const streakBonus = config.streakBonus || 5;
  const timeBonus = config.timeBonus || false;
  const timeLimit = config.timeLimit || 10; // 每题默认10秒

  for (let i = 0; i < answers.length; i++) {
    const isCorrect = isAnswerCorrect(answers[i], questions[i]);
    const timeSpent = answers[i].timeSpent || timeLimit;

    if (isCorrect) {
      streak++;
      let questionScore = baseScore;

      // 连击加成（每3连对加一次）
      if (streak >= 3) {
        questionScore += streakBonus * Math.floor(streak / 3);
      }

      // 时间奖励（提前交卷加分）
      if (timeBonus && timeSpent < timeLimit) {
        const timeLeftRatio = (timeLimit - timeSpent) / timeLimit;
        questionScore += Math.round(baseScore * 0.3 * timeLeftRatio);
      }

      totalScore += questionScore;

      if (streak > maxStreak) {
        maxStreak = streak;
      }
    } else {
      streak = 0; // 答错重置连对计数
    }
  }

  return {
    score: totalScore,
    maxStreak,
    correctCount: answers.filter((a, i) => isAnswerCorrect(a, questions[i])).length,
    total: answers.length
  };
}
```

### 14.2 策略模式：计分系统

```javascript
// 计分策略接口
class ScoringStrategy {
  calculate(answers, questions) {
    throw new Error('子类必须实现 calculate');
  }
}

// 考试型计分
class ExamScoringStrategy extends ScoringStrategy {
  calculate(answers, questions) {
    const correct = answers.filter(a => a.isCorrect).length;
    const score = Math.round((correct / answers.length) * 100);
    return { score, correctCount: correct, total: answers.length };
  }
}

// 游戏型计分
class GameScoringStrategy extends ScoringStrategy {
  constructor(config = {}) {
    super();
    this.baseScore = config.baseScore || 10;
    this.streakBonus = config.streakBonus || 5;
    this.timeBonus = config.timeBonus || false;
  }

  calculate(answers, questions) {
    let score = 0, streak = 0, maxStreak = 0;
    answers.forEach(a => {
      if (a.isCorrect) {
        streak++;
        score += this.baseScore + (streak >= 3 ? this.streakBonus * Math.floor(streak/3) : 0);
        maxStreak = Math.max(maxStreak, streak);
      } else {
        streak = 0;
      }
    });
    return { score, maxStreak, correctCount: answers.filter(a => a.isCorrect).length };
  }
}

// 使用
class ScoringContext {
  setStrategy(strategy) { this.strategy = strategy; }
  calculate(answers, questions) { return this.strategy.calculate(answers, questions); }
}
```

class QuizState {
  constructor(session) { this.session = session; }
  canSelect() { return false; }
  canSubmit() { return false; }
  canShowExplanation() { return false; }
}

// 答题中状态
class PlayingState extends QuizState {
  canSelect() { return true; }
  canShowExplanation() { 
    return this.session.config.feedbackMode === 'instant'; 
  }
  select(optionIndex) {
    this.session.recordAnswer(optionIndex);
    if (this.session.config.feedbackMode === 'instant') {
      this.session.judgeCurrent();
      this.session.setState(new FeedbackState(this.session));
    }
  }
}

// 等待下一题状态（即时模式）
class FeedbackState extends QuizState {
  canShowExplanation() { return true; }
  next() {
    if (this.session.isLast()) {
      this.session.setState(new SubmittedState(this.session));
    } else {
      this.session.next();
      this.session.setState(new PlayingState(this.session));
    }
  }
}

// 已提交状态
class SubmittedState extends QuizState {
  canSubmit() { return true; }
  canShowExplanation() { return true; }
}

// 答题会话（状态机）
class QuizSession {
  constructor(questions, config) {
    this.questions = questions;
    this.config = config;
    this.answers = questions.map(() => ({ userAnswer: null, isCorrect: null }));
    this.current = 0;
    this.state = new PlayingState(this);
  }
  
  setState(state) { this.state = state; }
  select(optionIndex) { this.state.select(optionIndex); }
  next() { this.state.next(); }
  submit() { this.state.submit(); }
}

      single_choice: SingleChoiceQuestion,
