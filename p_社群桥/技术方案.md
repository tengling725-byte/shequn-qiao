## 四、核心功能实现

### 4.1 用户认证

```typescript
// 微信登录
async function wechatLogin(code: string) {
  // 1. 通过code获取openid
  const appid = process.env.WECHAT_APPID;
  const secret = process.env.WECHAT_SECRET;
  const url = `https://api.weixin.qq.com/sns/oauth2/access_token?appid=${appid}&secret=${secret}&code=${code}&grant_type=authorization_code`;

  const response = await fetch(url);
  const data = await response.json();

  // 2. 查询或创建用户
  const user = await db.user.findUnique({
    where: { wechat_openid: data.openid }
  });

  if (!user) {
    return await db.user.create({
      data: { wechat_openid: data.openid }
    });
  }

  return user;
}
```

### 4.3 搜索功能

```typescript
// 搜索资源/需求
async function searchResources(params: {
  communityId: string;
  keyword?: string;
  type?: string;
  category?: string;
  tags?: string[];
  page: number;
  pageSize: number;
}) {
  const where: any = {
    community_id: params.communityId,
    is_public: true
  };

  if (params.keyword) {
    where.OR = [
      { title: { contains: params.keyword } },
      { description: { contains: params.keyword } },
      { tags: { has: params.keyword } }
    ];
  }

  if (params.type) {
    where.type = params.type;
  }

  if (params.category) {
    where.category = params.category;
  }

  return await db.resource.findMany({
    where,
    skip: (params.page - 1) * params.pageSize,
    take: params.pageSize,
    orderBy: { created_at: 'desc' }
  });
}
```

### 4.4 智能推荐

```typescript
// 基于用户的资源和需求进行匹配推荐
async function getRecommendations(userId: string, communityId: string) {
  // 1. 获取用户发布的资源和需求
  const userResources = await db.resource.findMany({
    where: { user_id: userId, community_id: communityId }
  });

  const userNeeds = await db.need.findMany({
    where: { user_id: userId, community_id: communityId }
  });

  // 2. 获取可能匹配的资源/需求
  const recommendations = [];

  // 为需求推荐资源
  for (const need of userNeeds) {
    const matchedResources = await db.resource.findMany({
      where: {
        community_id: communityId,
        is_public: true,
        category: need.category,
        NOT: { user_id: userId }
      },
      take: 5
    });

    recommendations.push({
      type: 'resource',
      matchedTo: need.id,
      items: matchedResources
    });
  }

  // 为资源推荐需求
  for (const resource of userResources) {
    const matchedNeeds = await db.need.findMany({
      where: {
        community_id: communityId,
        is_public: true,
        category: resource.category,
        NOT: { user_id: userId }
      },
      take: 5
    });

    recommendations.push({
      type: 'need',
      matchedTo: resource.id,
      items: matchedNeeds
    });
  }

  return recommendations;
}
```

async function sendMatchNotification(userId: string, matchInfo: any) {
  const user = await db.user.findUnique({
    where: { id: userId }

  if (!user.wechat_openid) return;
